def fmt(v):
    return f"${v/1000:.1f}K" if v >= 1000 else f"${v:.0f}"

def fmtFull(v):
    return f"${v:,.0f}"

# ── UI Components ─────────────────────────────────────────────────────────────

def kpi_card(label, value, sub=None, icon="☕", highlight=False):
    bg_style = f"background: linear-gradient(135deg, {AMBER}, {MOCHA});" if highlight else "background: rgba(28,10,0,0.6);"
    border_style = f"border: 1px solid {'transparent' if highlight else 'rgba(217,119,6,0.3)'};"
    label_color = "#FEF3C7" if highlight else "#D97706"
    value_color = "#fff" if highlight else "#FEF3C7"
    sub_color = "rgba(255,255,255,0.7)" if highlight else "rgba(254,243,199,0.5)"
    
    html = f"""
    <div style="{bg_style} {border_style} border-radius: 12px; padding: 18px 20px; backdrop-filter: blur(8px); flex: 1; min-width: 140px; margin-bottom: 10px;">
        <div style="font-size: 22px; margin-bottom: 6px;">{icon}</div>
        <div style="color: {label_color}; font-size: 11px; font-weight: 700; letter-spacing: 2px; text-transform: uppercase; margin-bottom: 4px;">{label}</div>
        <div style="color: {value_color}; font-size: 26px; font-weight: 800; line-height: 1;">{value}</div>
        {f'<div style="color: {sub_color}; font-size: 11px; margin-top: 4px;">{sub}</div>' if sub else ''}
    </div>
    """
    return html

def section_title(children, sub=None):
    html = f"""
    <div style="margin-bottom: 20px;">
        <h2 style="color: #FEF3C7; font-size: 18px; font-weight: 800; letter-spacing: 1px; margin: 0; font-family: 'Georgia', serif;">{children}</h2>
        {f'<p style="color: rgba(200,149,108,0.7); font-size: 12px; margin: 4px 0 0; font-style: italic;">{sub}</p>' if sub else ''}
        <div style="width: 40px; height: 2px; background: linear-gradient(90deg, #D97706, transparent); margin-top: 8px;"></div>
    </div>
    """
    return html

def insight_box(children, icon="☕"):
    html = f"""
    <div style="background: rgba(217,119,6,0.08); border: 1px solid rgba(217,119,6,0.2); border-left: 3px solid #D97706; border-radius: 8px; padding: 10px 14px; margin-top: 10px; display: flex; gap: 10px; align-items: flex-start;">
        <span style="font-size: 16px; flex-shrink: 0;">{icon}</span>
        <p style="color: rgba(254,243,199,0.8); font-size: 12px; margin: 0; line-height: 1.6;">{children}</p>
    </div>
    """
    return html
